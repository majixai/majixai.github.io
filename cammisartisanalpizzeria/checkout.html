Connect a real endpoint to submit it.'; const receipt = document.createElement('div'); receipt.className = 'receipt'; const lines = order.items.map(item => `${item.qty} × ${item.name} — ${money(item.price * item.qty)}`); lines.push('', `${order.fulfillment === 'delivery' ? 'Delivery' : 'Pickup'} · ${order.name}`, `Estimated total: ${money(order.total)}`, `Payment: ${order.payment}`); if (order.fulfillment === 'delivery') lines.push(`Address: ${order.address}, ${order.city}`); if (order.notes) lines.push(`Notes: ${order.notes}`); if (backendResult && !backendResult.ok) { lines.push('', `Status: ${backendResult.message || 'Not submitted'}`); } receipt.textContent = lines.join('\n'); receipt.style.whiteSpace = 'pre-line'; const disclaimer = document.createElement('p'); disclaimer.textContent = backendResult && backendResult.ok ?
